Ubuntu / Debian

echo "deb [signed-by=/usr/share/keyrings/yeager-keyring.gpg] https://yeager.github.io/debian-repo stable main" \
  | sudo tee /etc/apt/sources.list.d/yeager.list
curl -fsSL https://yeager.github.io/debian-repo/yeager-keyring.gpg \
  | sudo tee /usr/share/keyrings/yeager-keyring.gpg > /dev/null
sudo apt update
sudo apt install bildordbok
